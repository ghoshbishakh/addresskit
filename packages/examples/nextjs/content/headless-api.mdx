# Headless Hooks

Build fully custom address form interfaces without any pre-built markup using AddressKit headless hooks.

## `useAddressForm`

The primary hook for managing form state, dynamic schemas, country cascading, and validation:

```tsx
import { useAddressForm } from "@addresskit/react";

function CustomForm() {
  const {
    address,
    fields,
    errors,
    countries,
    isValid,
    isValidating,
    setCountry,
    setFieldValue,
    setAddress,
    reset,
  } = useAddressForm({
    initialAddress: { country: "US" },
    allowedCountries: ["US", "CA", "GB"],
  });

  return (
    <form>
      <div>
        <label htmlFor="country">Country</label>
        <select
          id="country"
          value={address.country ?? ""}
          onChange={(e) => setCountry(e.target.value)}
        >
          {countries.map((c) => (
            <option key={c.code} value={c.code}>
              {c.name}
            </option>
          ))}
        </select>
      </div>

      {fields.map((field) => (
        <div key={field.id}>
          <label htmlFor={field.id}>{field.label}</label>
          <input
            id={field.id}
            value={address[field.id] ?? ""}
            placeholder={field.placeholder}
            onChange={(e) => setFieldValue(field.id, e.target.value)}
          />
          {errors[field.id] && <span role="alert">{errors[field.id]}</span>}
        </div>
      ))}
    </form>
  );
}
```

## `useCountries`

Fetch the list of supported countries with optional filtering:

```tsx
import { useCountries } from "@addresskit/react";

function CountryList() {
  const countries = useCountries(["US", "CA", "GB", "DE"]);

  return (
    <select>
      {countries.map((c) => (
        <option key={c.code} value={c.code}>
          {c.name}
        </option>
      ))}
    </select>
  );
}
```

## `useStates` and `useCities`

Load subregions on demand:

```tsx
import { useStates, useCities } from "@addresskit/react";

function Subregions() {
  const states = useStates("US");
  const cities = useCities("US", "CA");

  return (
    <div>
      <p>States in US: {states.length}</p>
      <p>Cities in California: {cities.length}</p>
    </div>
  );
}
```

## `useAddressValidation`

Validate address values in real time:

```tsx
import { useAddressValidation } from "@addresskit/react";

function ValidationBadge({ address }) {
  const { valid, errors } = useAddressValidation(address);

  if (valid) return <span>Address is valid</span>;
  return <span>{errors.length} validation errors</span>;
}
```

## `useAddressFormat`

Format an address into postal layout:

```tsx
import { useAddressFormat } from "@addresskit/react";

function AddressPreview({ address }) {
  const formatted = useAddressFormat(address);
  return <pre>{formatted}</pre>;
}
```
