# React Hook Form

AddressKit integrates with React Hook Form using the `@addresskit/react-hook-form` package.

## Installation

```bash
npm install @addresskit/react-hook-form @addresskit/react react-hook-form
```

## `<AddressController>`

The `<AddressController>` wraps `<Address>` as a controlled form field in React Hook Form:

```tsx
import { useForm } from "react-hook-form";
import { AddressController, addressValidationRule } from "@addresskit/react-hook-form";
import type { Address } from "@addresskit/core";

interface CheckoutFormData {
  name: string;
  email: string;
  shippingAddress: Partial<Address>;
}

export function CheckoutForm() {
  const {
    control,
    handleSubmit,
    formState: { errors },
  } = useForm<CheckoutFormData>({
    defaultValues: {
      shippingAddress: { country: "US" },
    },
  });

  const onSubmit = (data: CheckoutFormData) => {
    console.log("Submitted data:", data);
  };

  return (
    <form onSubmit={handleSubmit(onSubmit)} className="space-y-4 max-w-md mx-auto">
      <div>
        <label className="block text-sm font-medium">Shipping Address</label>
        <AddressController
          name="shippingAddress"
          control={control}
          rules={{
            validate: addressValidationRule(),
          }}
        />
        {errors.shippingAddress && (
          <p className="text-red-500 text-sm mt-1">
            {errors.shippingAddress.message as string}
          </p>
        )}
      </div>

      <button type="submit" className="px-4 py-2 bg-blue-600 text-white rounded">
        Place Order
      </button>
    </form>
  );
}
```

## `addressValidationRule`

`addressValidationRule(provider?, options?)` executes AddressKit address validation when React Hook Form validates the form. If any required field or postal regex fails, it returns the first error message to React Hook Form.
